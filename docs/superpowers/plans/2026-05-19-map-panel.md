```javascript
function triggerShipJump(colorId) {

  if (!pt) return;
  // 若船正在飛行中，直接覆蓋目標（優先最新合成）
  mapShip.sx           = mapShip.x;
  mapShip.sy           = mapShip.y;
  mapShip.tx           = pt.x;
  mapShip.ty           = pt.y;
  mapShip.frame        = 0;
  mapShip.totalFrames  = MAP_FLY_FRAMES;
  mapShip.flying       = true;
  mapShip.pauseTimer   = 0;
  mapShip.targetColorId = colorId;
  mapShip.returnPending = mapFlags[colorId]; // 已插旗 → 落地後回家
}
```

```javascript
function updateMapPanel() {
  // 更新濺起粒子
  mapSplash = mapSplash.filter(s => {
    s.frame++; s.x += s.vx; s.y += s.vy;
    return s.frame < s.maxFrame;
  });

  if (mapShip.flying) {
    mapShip.frame++;
    const t = Math.min(mapShip.frame / mapShip.totalFrames, 1);

    if (t >= 1) {
      // 落地
      mapShip.x = mapShip.tx;
      mapShip.y = mapShip.ty;
      mapShip.flying = false;

      // 濺起粒子（5 個）
      const splashColor = mapShip.targetColorId >= 0
        ? COLOR_HEX[mapShip.targetColorId] : '#ffffff';
      for (let i = 0; i < 5; i++) {
        const angle = (i / 5) * Math.PI * 2;
        mapSplash.push({
          x: mapShip.x, y: mapShip.y,
          vx: Math.cos(angle) * 1.8, vy: Math.sin(angle) * 1.8,
          frame: 0, maxFrame: 10, color: splashColor,
        });
      }

      // 首次到達 → 插旗
      if (mapShip.targetColorId >= 0 && !mapFlags[mapShip.targetColorId]) {
        mapFlags[mapShip.targetColorId] = true;
      }

      mapShip.pauseTimer = MAP_PAUSE_FRAMES;

    } else {
      // 拋物線插值
      const ease = t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t; // easeInOut
      mapShip.x = mapShip.sx + (mapShip.tx - mapShip.sx) * ease;
      const arcH = Math.abs(mapShip.tx - mapShip.sx) * 0.4 + 30;
      mapShip.y = mapShip.sy + (mapShip.ty - mapShip.sy) * ease
                  - arcH * 4 * t * (1 - t);
    }

  } else if (mapShip.pauseTimer > 0) {
    mapShip.pauseTimer--;
    if (mapShip.pauseTimer === 0 && mapShip.returnPending) {
      // 已插旗目標，飛回出發港
      mapShip.sx = mapShip.x; mapShip.sy = mapShip.y;
      mapShip.tx = MAP_HOME_X; mapShip.ty = MAP_HOME_Y;
      mapShip.frame = 0; mapShip.totalFrames = MAP_FLY_FRAMES;
      mapShip.flying = true;
      mapShip.returnPending = false;
      mapShip.targetColorId = -1;
    }
  }
}
```

```javascript
function drawMapPanel(ctx) {
  const px = MAP_PX, py = MAP_PY, pw = MAP_PW, ph = MAP_PH;
  const allLit = synthesizedColors.size >= 4;

  ctx.save();

  // ── 背景：深色漸層 + 像素格線 ──
  const grad = ctx.createLinearGradient(px, py, px + pw, py + ph);
  grad.addColorStop(0, '#0a1628');
  grad.addColorStop(1, '#0d2818');
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.roundRect(px, py, pw, ph, 8);
  ctx.fill();

  ctx.save();
  ctx.beginPath();
  ctx.roundRect(px, py, pw, ph, 8);
  ctx.clip();
  ctx.strokeStyle = 'rgba(255,255,255,0.05)';
  ctx.lineWidth = 1;
  for (let gx = px; gx <= px + pw; gx += 20) {
    ctx.beginPath(); ctx.moveTo(gx, py); ctx.lineTo(gx, py + ph); ctx.stroke();
  }
  for (let gy = py; gy <= py + ph; gy += 20) {
    ctx.beginPath(); ctx.moveTo(px, gy); ctx.lineTo(px + pw, gy); ctx.stroke();
  }
  ctx.restore();

  // ── 外框 ──
  if (allLit) {
    ctx.shadowColor = '#ffd700'; ctx.shadowBlur = 20;
    ctx.strokeStyle = '#ffd700'; ctx.lineWidth = 2;
  } else {
    ctx.strokeStyle = 'rgba(255,255,255,0.15)'; ctx.lineWidth = 1;
  }
  ctx.beginPath(); ctx.roundRect(px, py, pw, ph, 8); ctx.stroke();
  ctx.shadowBlur = 0;

  // ── 地形目標點 ──
  for (const pt of MAP_POINTS) {
    const lit = synthesizedColors.has(pt.colorId);

    // 發光圈（已合成）
    if (lit) {
      ctx.save();
      ctx.shadowColor = COLOR_HEX[pt.colorId];
      ctx.shadowBlur  = 18;
      ctx.fillStyle   = COLOR_HEX[pt.colorId] + '30';
      ctx.beginPath();
      ctx.arc(pt.x, pt.y, 22, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }

    // 地點圓圈
    ctx.fillStyle   = lit ? COLOR_HEX[pt.colorId] + 'bb' : '#1e293b';
    ctx.strokeStyle = lit ? COLOR_HEX[pt.colorId]        : '#334155';
    ctx.lineWidth   = lit ? 2 : 1;
    ctx.beginPath();
    ctx.arc(pt.x, pt.y, 18, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    // 地形圖示
    ctx.font         = '16px Arial';
    ctx.textAlign    = 'center';
    ctx.textBaseline = 'middle';
    ctx.globalAlpha  = lit ? 1 : 0.3;
    ctx.fillStyle    = '#fff';
    ctx.fillText(pt.icon, pt.x, pt.y);
    ctx.globalAlpha  = 1;

    // 旗子（已插旗）
    if (mapFlags[pt.colorId]) {
      const wave = Math.sin(Date.now() / 400) * 3;
      // 旗桿
      ctx.strokeStyle = '#d4a017';
      ctx.lineWidth   = 1.5;
      ctx.beginPath();
      ctx.moveTo(pt.x + 18, pt.y - 8);
      ctx.lineTo(pt.x + 18, pt.y - 30);
      ctx.stroke();
      // 旗面
      ctx.save();
      ctx.translate(pt.x + 18, pt.y - 30);
      ctx.rotate((wave * Math.PI) / 180);
      ctx.fillStyle = COLOR_HEX[pt.colorId];
      ctx.fillRect(0, 0, 13, 8);
      ctx.restore();
    }
  }

  // ── 濺起粒子 ──
  for (const s of mapSplash) {
    const progress = s.frame / s.maxFrame;
    ctx.globalAlpha = 1 - progress;
    ctx.fillStyle   = s.color;
    ctx.beginPath();
    ctx.arc(s.x, s.y, 2 + progress * 5, 0, Math.PI * 2);
    ctx.fill();
  }
  ctx.globalAlpha = 1;

  // ── 船 ──
  ctx.font         = '20px Arial';
  ctx.textAlign    = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText('⛵', mapShip.x, mapShip.y);

  // ── 出發港標籤 ──
  ctx.font      = '9px Arial';
  ctx.fillStyle = 'rgba(255,255,255,0.35)';
  ctx.textAlign = 'center';
  ctx.fillText('出發港', MAP_HOME_X, MAP_HOME_Y + 14);

  ctx.restore();
}
```
