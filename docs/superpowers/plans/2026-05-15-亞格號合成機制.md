```js
// === 合成機制 ===
function checkSynthesis() {
  let found = false;

  // ── Pass 1：掃描基礎合成（同色角色+武器水平相鄰 → 1×2）──
  for (let row = 0; row < ROWS; row++) {
    for (let col = 0; col < COLS - 1; col++) {
      const a = board[col][row];
      const b = board[col + 1][row];
      // 跳過已合成格
      if (a.isMerged || b.isMerged) continue;
      if (a.id < 0 || b.id < 0) continue;

      const aColorId = Math.floor(a.id / 2);
      const bColorId = Math.floor(b.id / 2);
      const aIsChar  = (a.id % 2 === 0);
      const bIsChar  = (b.id % 2 === 0);

      // 同色 且 一個是角色、一個是武器
      if (aColorId === bColorId && aIsChar !== bIsChar) {
        const colorId = aColorId;

        // 更新 splitCount
        if (synthesizedColors.has(colorId)) {
          splitCount[colorId]++;
        } else {
          splitCount[colorId] = 2;
          synthesizedColors.add(colorId);
        }

        // 標記盤面格
        const mergeId = mergedCells.length;
        board[col][row]     = { id: -1, isMerged: true, mergeId, colorId, mergeSize: 2, mergeCol: col, mergeRow: row, isMergeSlave: false, isMatched: false, yOffset: 0, scale: 1 };
        board[col+1][row]   = { id: -1, isMerged: true, mergeId, colorId, mergeSize: 2, mergeCol: col, mergeRow: row, isMergeSlave: true,  isMatched: false, yOffset: 0, scale: 1 };

        mergedCells.push({ col, row, colorId, size: 2 });
        found = true;
      }
    }
  }

  // ── Pass 2：掃描進階合成（兩個相鄰 1×2 不同色 → 1×4）──
  for (let i = 0; i < mergedCells.length; i++) {
    const mA = mergedCells[i];
    if (mA.size !== 2 || mA.upgraded) continue;
    for (let j = i + 1; j < mergedCells.length; j++) {
      const mB = mergedCells[j];
      if (mB.size !== 2 || mB.upgraded) continue;
      if (mA.row !== mB.row) continue;
      if (mA.colorId === mB.colorId) continue;

      // 相鄰：mA 右邊緊接 mB
      const leftM  = mA.col < mB.col ? mA : mB;
      const rightM = mA.col < mB.col ? mB : mA;
      if (leftM.col + 2 !== rightM.col) continue;

      // 進階合成
      splitCount[leftM.colorId]++;
      splitCount[rightM.colorId]++;

      const newMerge = { col: leftM.col, row: leftM.row, colorId: leftM.colorId, colorId2: rightM.colorId, size: 4 };
      leftM.upgraded = true;
      rightM.upgraded = true;
      mergedCells.push(newMerge);

      // 更新四個格
      for (let c = leftM.col; c < leftM.col + 4; c++) {
        board[c][leftM.row] = {
          id: -1, isMerged: true, mergeId: mergedCells.length - 1,
          colorId: leftM.colorId, colorId2: rightM.colorId,
          mergeSize: 4, mergeCol: leftM.col, mergeRow: leftM.row,
          isMergeSlave: (c !== leftM.col), isMatched: false, yOffset: 0, scale: 1
        };
      }
      found = true;
    }
  }

  // 過濾掉已 upgrade 的舊 1×2 記錄
  mergedCells = mergedCells.filter(m => !m.upgraded);
  return found;
}
```

```js
function calcWin() {
  let totalWin = 0;
  winLines = [];

  for (let symId = 0; symId < 8; symId++) {
    const colorId = Math.floor(symId / 2);
    const split   = splitCount[colorId];

    // 各列有效計數
    const colCounts = [];
    for (let col = 0; col < COLS; col++) {
      let count = 0;
      for (let row = 0; row < ROWS; row++) {
        const cell = board[col][row];
        if (!cell.isMerged && cell.id === symId) {
          count++;
        } else if (cell.isMerged && !cell.isMergeSlave) {
          // 合成符替代同色的角色與武器
          const cids = cell.colorId2 !== undefined
            ? [cell.colorId, cell.colorId2]
            : [cell.colorId];
          if (cids.includes(colorId)) count++;
        }
      }
      colCounts.push(count * split);
    }

    // 左→右連續列計 Ways
    let matchCount = 0;
    let ways = 1;
    for (let col = 0; col < COLS; col++) {
      if (colCounts[col] === 0) break;
      matchCount++;
      ways *= colCounts[col];
    }

    if (matchCount >= 3) {
      const rate = PAYOUTS[symId][matchCount - 3];
      const symWin = Math.floor(betAmount * rate * ways);
      totalWin += symWin;
      // 標記中獎格（用於消除動畫）
      for (let col = 0; col < matchCount; col++) {
        for (let row = 0; row < ROWS; row++) {
          const cell = board[col][row];
          if (cell.id === symId || (cell.isMerged && !cell.isMergeSlave)) {
            cell.isMatched = true;
          }
        }
      }
      winLines.push({ symId, matchCount, ways, win: symWin });
    }
  }

  return totalWin;
}
```

```js
function computeCurrentWays() {
  let total = 0;
  for (let symId = 0; symId < 8; symId++) {
    const colorId = Math.floor(symId / 2);
    const split = splitCount[colorId];
    let ways = 1;
    let valid = true;
    for (let col = 0; col < COLS; col++) {
      let count = 0;
      for (let row = 0; row < ROWS; row++) {
        const cell = board[col][row];
        if (!cell.isMerged && cell.id === symId) count++;
        else if (cell.isMerged && !cell.isMergeSlave) {
          const cids = cell.colorId2 !== undefined ? [cell.colorId, cell.colorId2] : [cell.colorId];
          if (cids.includes(colorId)) count++;
        }
      }
      if (count === 0) { valid = false; break; }
      ways *= count * split;
    }
    if (valid) total += ways;
  }
  return total || 1024;
}
```
